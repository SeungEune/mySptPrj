* 목록 조회 js
document.addEventListener('DOMContentLoaded', (event) => {
    fn_init();

    /**
     * 검색 버튼 클릭 이벤트
     */
    let searchBtnEl = document.getElementById('searchBtn');
    let searchKeywordEl = document.getElementById('searchKeyword');

    searchBtnEl.addEventListener('click', () => {
        fn_searchData(1);
    });
    searchKeywordEl.addEventListener('keydown', function(event) {
        if (event.key === 'Enter')  fn_searchData(1);
    });

    let droneDistListContsEl = document.getElementById('droneDistListConts');
    droneDistListContsEl.addEventListener('click', (e) => {
        //촬영지구 등록
        if(e.target && e.target.matches('button#addProjectBtn')){
            callModule.post(Util.getRequestUrl('/drone/plan/detail.do'), null, 'GET');
        }
    })

    /**
     * 날짜 선택 벨리데이션
     */
    let bgndeDateEl = document.getElementById('bgnde');
    let enddeDateEl = document.getElementById('endde');
    bgndeDateEl.addEventListener('change', function(e){
        gf_setMinDate(e, enddeDateEl);
    });

    /**
     * 상세 보기 화면 이동
     */
    droneDistListContsEl.addEventListener('click', (e) => {
        let target = e.target;
        while (target && target !== droneDistListContsEl) {
            if (target.matches('tr[data-project-sn]')) {
                let statusCode = target.dataset.statusCode;
                let assetTyCode = target.dataset.assettyCode;
                let parsingStatus = statusCode.replace("DPS", "").replaceAll("0","");
                let statusNum =  parseInt(parsingStatus);
                let nextStatusCode;

                switch(assetTyCode){
                    case 'PAT001' :
                        nextStatusCode ="DPS" + String(statusNum === 8 ? 8 : statusNum + 1).padStart(3, "0");
                        break;
                    case 'PAT002' :
                        if(statusNum < 2){
                            nextStatusCode ="DPS" + String(statusNum === 8 ? 8 : statusNum + 1).padStart(3, "0");
                        } else {
                            nextStatusCode = 'DPS006'
                        }
                        break;
                }

                let data = {
                    projectSn: target.dataset.projectSn
                }
                callModule.post(Util.getRequestUrl(pageInfoJson[nextStatusCode].url), data, pageInfoJson[nextStatusCode].method);
                return;
            }
            target = target.parentElement;
        }
    })
});

function fn_init(){
    let pageNo = document.getElementById('searchPageNo').value;
    fn_searchData(pageNo);
}

function fn_searchData(pageNo){
    let searchType = document.getElementById('searchType').value;
    let searchKeyword = document.getElementById('searchKeyword').value;

    let data = {
        projectSeCode :  document.getElementById('projectSeCode').value,
        bsnsTyCode : document.getElementById('bsnsTyCode').value,
        statusCode : document.getElementById('statusCode').value,
        qlityGoalGradCode : document.getElementById('qlityGoalGradCode').value,
        bgnde : document.getElementById('bgnde').value,
        endde : document.getElementById('endde').value,
        searchVO : {
            searchType : Util.isEmpty(searchType) ? '' : searchType,
            searchKeyword : Util.isEmpty(searchKeyword) ?  '' : searchKeyword,
            pageNo : Util.isEmpty(pageNo) ?  '1' : pageNo,
        }
    };

    const callback = function (result){
        $('#distListTd').empty();

        if(result.droneDistList != null && result.droneDistList.length > 0){
            $('#distListTd').append($('#tmplDistListTmpl').tmpl(result.droneDistList));
            let target = document.getElementById('paginationArea');
            setPagination(result.pagingVO, target, 'fn_searchData');
        } else {
            $('#distListTd').append($('#emptyDistListTmpl').tmpl(""));
        }
    }

    callModule.call(Util.getRequestUrl('/drone/list.do'), data, callback, true);
}
